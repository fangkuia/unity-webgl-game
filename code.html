// SquareController.cs - プレイヤーコントローラー / 玩家控制器
// WASD movement + jump + auto-bounce + custom gravity
public class SquareController : MonoBehaviour
{
    public float moveSpeed = 8f;           // 移動速度 / 移速
    public float jumpForce = 10f;          // ジャンプ力 / 跳跃力
    public float gravityScale = 2.5f;       // 上升重力 / 上升时重力倍率
    public float fallGravityScale = 5f;    // 下降重力 / 下落时重力倍率
    public bool autoBounce = true;         // 自动弹跳 / 自動バウンス
    private Rigidbody rb; private bool grounded;
    void Start() { rb = GetComponent<Rigidbody>(); }
    void Update() {
        float h = Input.GetAxis("Horizontal");
        rb.linearVelocity = new Vector3(h*moveSpeed, rb.linearVelocity.y, 0);
        if (grounded && Input.GetButtonDown("Jump")) {
            rb.linearVelocity = new Vector3(rb.linearVelocity.x, jumpForce, 0);
            grounded = false;
        }
        float g = rb.linearVelocity.y > 0 ? gravityScale : fallGravityScale;
        rb.AddForce(Physics.gravity * (g - 1f), ForceMode.Acceleration);
        if (!grounded) transform.Rotate(0,0,360*Time.deltaTime);
    }
    void OnCollisionEnter(Collision c) {
        foreach (var p in c.contacts)
            if (p.normal.y > 0.5f) {
                grounded = true;
                if (autoBounce) rb.linearVelocity = new Vector3(rb.linearVelocity.x, jumpForce*0.8f, 0);
            }
    }
}

PlatformGenerator プラットフォーム / 平台

// PlatformGenerator.cs - 程序化平台生成 / ランダム足場生成
// 随机生成平台、陷阱、斜坡 / 通常足場・トラップ・坂道
public class PlatformGenerator : MonoBehaviour
{
    public bool isInfiniteMode, isHardMode;
    public Transform floorTile, topTile, leftWall, rightWall, player;

    public int rowCount = 5;
    public int platformsPerRowMin = 2, platformsPerRowMax = 2;
    public float platformWidthMin = 2f, platformWidthMax = 5f;
    public Color platformColor;

    public float trapChance = 0.15f;       // 陷阱概率 / トラップ確率
    public float trapDestroyDelay = 0.3f;  // 销毁延迟 / 破壊遅延
    public Color trapColor;
    public float slopeChance = 0.2f;       // 斜坡概率 / 坂道確率
    public Color slopeColor;
    public int generationRange = 10;       // 生成范围(向上) / 生成範囲(上)

    public void Generate() {
        ClearPlatforms();
        Shader s = Shader.Find("Universal Render Pipeline/Unlit");
        for (int r=0; r<rowCount; r++) {
            float y = Mathf.Lerp(floorY,topY,(float)(r+1)/(rowCount+1));
            CreateRow(y, r, Random.Range(platformsPerRowMin, platformsPerRowMax+1));
        }
    }
    GameObject CreatePlatform(float x,float y,float w,Material m) {
        var p = GameObject.CreatePrimitive(PrimitiveType.Cube);
        p.transform.SetParent(transform);
        p.transform.localPosition = new Vector3(x,y,0);
        p.transform.localScale = new Vector3(w,0.3f,2f);
        p.GetComponent<MeshRenderer>().sharedMaterial = m;
        return p;
    }
    Material MakeMaterial(Shader s, Color c) {
        var m = new Material(s); m.color = c; return m;
    }
}

CameraFollow + GoalTrigger + TrapPlatform

// CameraFollow.cs - 相机跟随 / カメラ追従
public class CameraFollow : MonoBehaviour {
    public Transform target; public float smoothTime = 0.3f;
    public bool useBounds = true, infiniteMode; public float minX,maxX,minY,maxY;
    private Vector3 vel;
    void LateUpdate() {
        if (!target) return;
        var tp = target.position;
        if (useBounds) { tp.x=Mathf.Clamp(tp.x,minX,maxX); if(!infiniteMode) tp.y=Mathf.Clamp(tp.y,minY,maxY); }
        transform.position = Vector3.SmoothDamp(transform.position, new Vector3(tp.x,tp.y,transform.position.z), ref vel, smoothTime);
    }
}

}

WebGLBuilder Editor

// 自動ビルド: バージョン管理 + Shader保護 / 自动化构建: 版本递增 + Shader保护
using UnityEditor; using System.Text.RegularExpressions;
public static class WebGLBuilder {
    [MenuItem("Tools/Build WebGL")]
    public static void Build() {
        // ① Auto-increment version / 版本号+1
        string c = File.ReadAllText("Assets/Scripts/GameUI.cs");
        var m = Regex.Match(c, "VERSION\\s*=\\s*\"v(\\d+)\"");
        int v = int.Parse(m.Groups[1].Value) + 1;
        c = Regex.Replace(c, "VERSION\\s*=\\s*\"v\\d+\"", "VERSION = \"v"+v+"\"");
        File.WriteAllText("Assets/Scripts/GameUI.cs", c);

        // ② Only test.unity scene / 只包含test场景
        var ss = new List<EditorBuildSettingsScene> {
            new EditorBuildSettingsScene("Assets/test.unity", true) };
        EditorBuildSettings.scenes = ss.ToArray();

        // ③ Protect URP shaders / Shader防剥离
        AddShader("Universal Render Pipeline/Lit");
        AddShader("Universal Render Pipeline/Unlit");

        // ④ Build / 执行构建
        PlayerSettings.WebGL.memorySize = 256;
        PlayerSettings.WebGL.compressionFormat = WebGLCompressionFormat.Disabled;
        string p = Path.Combine(Application.dataPath, "../Build/WebGL");
        BuildPipeline.BuildPlayer(ss.ToArray(), p, BuildTarget.WebGL, BuildOptions.None);
    }
    static void AddShader(string n) {
        var s = Shader.Find(n); if (!s) return;
        var g = new SerializedObject(AssetDatabase.LoadAssetAtPath<Object>(
            "ProjectSettings/GraphicsSettings.asset"));
        var p = g.FindProperty("m_AlwaysIncludedShaders");
        int i = p.arraySize; p.InsertArrayElementAtIndex(i);
        p.GetArrayElementAtIndex(i).objectReferenceValue = s;
        g.ApplyModifiedPropertiesWithoutUndo();
    }
}
